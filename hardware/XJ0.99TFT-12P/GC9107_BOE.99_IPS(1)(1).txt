void Init_LCD(void)
{
	//-------- Reset Sequence-----//
	LCD_nRST = 1;
	delayms(50);
	LCD_nRST = 0;
	delayms(50);
	LCD_nRST = 1;
	delayms(120);
	//------end Reset Sequence-----//


	LCD_CtrlWrite(0xFE);			
	LCD_CtrlWrite(0xEF);
		
	LCD_CtrlWrite(0xB0);			
	LCD_DataWrite(0xC0); 
		
	LCD_CtrlWrite(0xB2);

	LCD_CtrlWrite(0x3A);		
	LCD_DataWrite(0x05);

void Lcd_enter_sleep(void)
{
	LCD_CtrlWrite(0xfe);
	LCD_CtrlWrite(0xef);
	LCD_CtrlWrite(0x28);
	delayms(120);
	LCD_CtrlWrite(0x10);
	delayms(50);
}

void Lcd_exit_sleep(void)
{
	LCD_CtrlWrite(0xfe);
	LCD_CtrlWrite(0xef);
	LCD_CtrlWrite(0x11);
	delayms(120) ;
	LCD_CtrlWrite(0x29);
}
